// AI Detector Lite - Fixed allow drop + paste const input = document.getElementById('inputText') || document.getElementById('essay'); const dropZone = document.getElementById('dropZone') || input; const fileInput = document.getElementById('fileInput'); const resultDiv = document.getElementById('heuristicResult') || document.getElementById('result'); function heuristicCheck(text){ const t = text.toLowerCase(); let aiScore = 0; const patterns = [ /as an ai language model/i, /in conclusion,.*overall/i, /delve into/i, /moreover|furthermore|additionally/i, /it is important to note/i, /in today's rapidly evolving world/i ]; let found = []; patterns.forEach(p=>{ if(p.test(text)){ aiScore+=15; found.push(p.source); } }); // Repetitive sentence length = AI pattern const sentences = text.split(/[.!?]+/).filter(s=>s.trim().length>10); const avgLen = sentences.reduce((a,s)=>a+s.split(/\s+/).length,0)/ (sentences.length||1); if(avgLen>22 && avgLen<28) aiScore+=20; const humanPct = Math.max(5, 100 - aiScore); return {humanPct, found, textLen: text.length}; } // Allow drop PDF/DOCX/TXT into detector function setupDrop(){ if(!dropZone) return; ['dragenter','dragover'].forEach(ev=>{ dropZone.addEventListener(ev, e=>{ e.preventDefault(); dropZone.style.outline='3px dashed #8b5cf6'; dropZone.style.background='rgba(139,92,246,0.06)'; }); }); ['dragleave','drop'].forEach(ev=>{ dropZone.addEventListener(ev, e=>{ e.preventDefault(); dropZone.style.outline=''; dropZone.style.background=''; }); }); dropZone.addEventListener('drop', async e=>{ e.preventDefault(); const file = e.dataTransfer.files[0]; if(!file) return; await handleFile(file); }); } async function handleFile(file){ const buf = await file.arrayBuffer(); let txt=''; try{ if(file.name.endsWith('.pdf')){ const pdf = await pdfjsLib.getDocument({data:buf}).promise; for(let i=1;i<=pdf.numPages;i++){ const p=await pdf.getPage(i); const tc=await p.getTextContent(); txt+=tc.items.map(it=>it.str).join(' ')+' '; } } else if(file.name.endsWith('.docx')){ const r = await mammoth.extractRawText({arrayBuffer:buf}); txt=r.value; } else { txt = await file.text(); } if(input) input.value = txt; console.log('[Aptumi] AI Detector - parsed locally, no upload, len', txt.length); runCheck(); }catch(err){ alert('Gagal parse: '+err.message); } } function runCheck(){ const text = (input?.value||'').trim(); if(!text){ alert('Paste atau drop essay dulu'); return; } if(text.length<50){ alert('Text terlalu pendek, min 50 char'); return; } const res = heuristicCheck(text); if(resultDiv){ resultDiv.innerHTML = 'Likely '+(res.humanPct>60?'human-written':'AI-like')+' ('+res.humanPct+'% human pattern)
' + '• No strong AI pattern found: '+(res.found.length? res.found.join(', ') : 'clean') + '
Length: '+res.textLen+' chars - This is offline heuristic, not GPTZero, no data sent'; } } document.getElementById('checkBtn')?.addEventListener('click', runCheck); if(fileInput){ document.getElementById('browseBtn')?.addEventListener('click', ()=>fileInput.click()); fileInput.addEventListener('change', ()=>{ if(fileInput.files[0]) handleFile(fileInput.files[0]); }); } setupDrop(); console.log('[Aptumi] AI Detector Lite fixed - drop enabled');
