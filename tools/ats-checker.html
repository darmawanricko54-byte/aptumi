// ATS Resume Checker - Fixed Drop const dropZone = document.getElementById('dropZone') || document.querySelector('.drop-zone') || document.body; const fileInput = document.getElementById('fileInput'); const browseBtn = document.getElementById('browseBtn'); const jobDescInput = document.getElementById('jobDesc'); const resultDiv = document.getElementById('result'); function showAlert(msg){ console.log('[Aptumi] '+msg); } function highlight(){ dropZone.style.outline='3px dashed #7c3aed'; dropZone.style.background='rgba(124,58,237,0.06)'; } function unhighlight(){ dropZone.style.outline=''; dropZone.style.background=''; } ['dragenter','dragover'].forEach(ev=>{ dropZone.addEventListener(ev, e=>{ e.preventDefault(); e.stopPropagation(); highlight(); }); }); ['dragleave','drop'].forEach(ev=>{ dropZone.addEventListener(ev, e=>{ e.preventDefault(); e.stopPropagation(); unhighlight(); }); }); dropZone.addEventListener('drop', async (e)=>{ const file = e.dataTransfer.files[0]; if(!file){ showAlert('No file dropped'); return; } await handleFile(file); }); if(browseBtn && fileInput){ browseBtn.addEventListener('click', ()=> fileInput.click()); fileInput.addEventListener('change', async ()=>{ if(fileInput.files[0]) await handleFile(fileInput.files[0]); }); } async function handleFile(file){ if(file.size > 5*1024*1024){ alert('Max 5MB'); return; } const buf = await file.arrayBuffer(); let text = ''; try{ if(file.name.endsWith('.pdf')){ const pdf = await pdfjsLib.getDocument({data: buf}).promise; for(let i=1;i<=pdf.numPages;i++){ const page = await pdf.getPage(i); const tc = await page.getTextContent(); text += tc.items.map(it=>it.str).join(' ')+'\n'; } } else if(file.name.endsWith('.docx')){ const result = await mammoth.extractRawText({arrayBuffer: buf}); text = result.value; } if(!text.trim()){ alert('File kosong / tidak terbaca'); return; } window.__aptumi_resume_text = text; console.log('[Aptumi] Parsed locally, no upload. Length:', text.length); scoreATS(text, jobDescInput?.value||''); }catch(err){ console.error(err); alert('Gagal parse: '+err.message); } } function scoreATS(resume, jobDesc){ let score = 0; const fixes = []; if(/experience|pengalaman/i.test(resume)) score+=25; else fixes.push('Tambah Experience'); if(/education|pendidikan|university/i.test(resume)) score+=25; else fixes.push('Tambah Education'); if(/@|email/i.test(resume)) score+=15; else fixes.push('Tambah Email'); if(resume.length>300) score+=15; else fixes.push('Resume terlalu pendek'); if(jobDesc){ const keywords = jobDesc.toLowerCase().split(/\W+/).filter(w=>w.length>3); const match = keywords.filter(k=>resume.toLowerCase().includes(k)).length; score += Math.min(20, Math.round(match/keywords.length*20)); } else { score+=20; } document.getElementById('atsScore').innerText = score+'/100'; document.getElementById('fixes').innerText = fixes.join(', ') || 'Good to go!'; }
